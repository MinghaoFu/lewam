\begin{frame}{From VLA / BC to World--Action Models (WAM)}
\begin{itemize}
  \item VLA / BC: observation $\to$ action --- imitate, no explicit dynamics.
  \item \textbf{WAM}: jointly model world \emph{and} action (predict next state \emph{and} action) $\Rightarrow$ a forward model you can plan with / verify against, not just imitate.
  \item Community shift 2024$\to$26: explicit world dynamics over BC-only (the trend you flagged).
  \item \textbf{LeWAM} $=$ LeWM's JEPA latent $+$ a jointly-trained action pathway. ``All-in, all-out'' state--action; the \emph{latent} WAM in your Le-family.
\end{itemize}
\end{frame}

\begin{frame}{The WAM field is going \emph{search-free} --- LeWAM asks ``should we still plan?''}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\textbf{Search-free WAMs (the trend)}
\begin{itemize}\footnotesize
  \item \textbf{DreamZero}: video diffusion, $p(\text{video},a)$, physics from internet-scale video.
  \item \textbf{Being-H0.7}: a \emph{latent} WAM (our exact term!), 200k h egocentric video, future-aware latent workspace.
  \item Both \emph{amortize planning into one forward pass} --- no test-time search / rollout.
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\textbf{LeWAM (contrarian)}
\begin{itemize}\footnotesize
  \item Light JEPA latent ($\sim$$25$\,GB), joint $(z_{t+1},a_t)$.
  \item \emph{Keeps} a short-horizon search loop + history/past-action policy.
  \item Asks: \emph{when is search worth keeping?}
\end{itemize}
\end{column}
\end{columns}
\vspace{0.4em}
\textbf{Positioning.} LeWAM is \emph{not} ``the latent WAM'' (Being-H0.7 owns that at scale). It is the \textbf{per-family analysis} of when latent WM-planning pays off (geometry: yes; contact: no $\Rightarrow$ fall back to history-BC) --- the question the search-free WAMs sidestep by never searching.
\end{frame}

\begin{frame}{LeWAM architecture (concise)}
\begin{itemize}
  \item Encoder (DINOv2/ViT) $\to$ latent $z$; WM predictor for $z$-dynamics, SIGReg-regularized $\Rightarrow$ isotropic, plannable.
  \item Action / intention head: autoregressive on latent history $z_{\le t}$ + past actions $a_{<t}$ $\to$ action-embedding $\to$ decoder $\to$ raw action.
  \item Frozen-CLIP task token (multitask); padded+masked action dims (cross-embodiment).
  \item One latent, three eval modes: history-BC \,/\, intuition-guided \,/\, CEM-planning.
\end{itemize}
\end{frame}

\begin{frame}{Finding 1 --- planning is \emph{per-family}}
\begin{itemize}
  \item WM-planning (CEM) \textbf{works} on geometry / goal-reaching: OGBench-Cube $68\!\to\!78\!\to\!88$, Push-T, Reacher, Two-Room.
  \item \textbf{Collapses to $\approx 0$} on contact-rich robomimic Lift/Can/Square.
  \item \textbf{Mechanism} (verified): both are grasp tasks, identical eval --- so \emph{not} ``push vs grasp.'' On Cube the goal-image shows the object \emph{displaced} $\Rightarrow$ CEM must grasp+carry to lower latent cost. On Lift the goal is the cube $+$few cm, \emph{swamped by arm pose} $\Rightarrow$ CEM matches pose, skips the grasp; grasp only weakly exposed in one frame (probe $0.82\!\to\!0.99$ with history). On \textbf{Can/Square} the latent \emph{does} see grasp ($0.96$/$0.999$) but the task is \emph{multi-stage} (approach$\to$grasp$\to$place) so flat CEM still fails. (Data: cube $\sim$1M vs robomimic 200 demos.)
\end{itemize}
\end{frame}

\begin{frame}{Finding 2 --- history-BC $\gg$ CEM on contact}
\begin{itemize}
  \item On robomimic, history / past-action conditioned BC \textbf{recovers grasp state}: post-grasp $P(\text{success}\,|\,\text{grasp})\ 76\!\to\!99$.
  \item history-BC: Lift $\sim$95, Can $\sim$48--72, Square $\sim$60s\ \ vs\ \ online CEM $=0$.
  \item The action pathway, conditioned on \emph{past actions}, carries contact tasks where single-frame WM planning cannot.
  \item Closest published latent planner LDP $0.69/0.70/0.46$; we beat it on Square.
\end{itemize}
\end{frame}

\begin{frame}{Finding 3 (honest negative) --- the action-head A/B is within noise}
\begin{itemize}
  \item detach=TRUE / FALSE / solveact (raw action) / SIGReg-on-act-emb: all \textbf{tied $\approx 72\pm5$ on Can}, $N{=}50$, 3 \emph{training} seeds.
  \item The $\sim$10-pt \textbf{training-run variance swamps} the 4--7-pt effects single runs showed.
  \item Lesson: multi-\emph{train}-seed is mandatory; single-run A/Bs (and eval-seed-only ``3-seed'') mislead. We caught a false ``solveact wins / intent\_loss hurts'' headline before publishing.
\end{itemize}
\end{frame}

\begin{frame}{Contribution \& positioning}
\begin{itemize}
  \item LeWAM $=$ unified JEPA \emph{latent} world-action model. Contributions:
  \begin{enumerate}
    \item the per-family \emph{``when does WM-planning help''} analysis;
    \item history / past-action-conditioned action pathway recovering contact state (history-BC $\gg$ CEM);
    \item the WAM reframe + honest variance accounting (the latent counterpart to DreamZero's video WAM).
  \end{enumerate}
  \item \textbf{Not} a fine action-head SR win --- that washed out. The story is mechanism + analysis + the history-conditioned wedge.
\end{itemize}
\end{frame}

\begin{frame}{Differentiation (related work, head-on)}
\footnotesize
\begin{itemize}
  \item \textbf{DreamZero} (video WAM, no search): LeWAM $=$ light latent + the planning analysis it sidesteps.
  \item \textbf{Newt / PiJEPA} (JEPA priors, Markovian/raw-action): LeWAM $=$ history+past-action conditioned.
  \item \textbf{GC-IDM} (amortize on LeWM latent --- must-beat): LeWAM $=$ warm-start-\emph{then}-refine, not replace search.
  \item \textbf{WorldDP / LDP} (hierarchical + diffusion exec): they \emph{assume} flat planning fails; we \emph{map when} it fails.
  \item \textbf{MR.Q} (repr.\ $>$ planning): our action pathway \emph{is} that predictive representation; we restrict planning to where it provably helps.
\end{itemize}
\end{frame}
